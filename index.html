# Single-cell RNA-seq Analysis

import scanpy as sc

adata = sc.read_10x_mtx('data/')
sc.pp.filter_cells(adata, min_genes=200)
sc.pp.normalize_total(adata)
sc.tl.umap(adata)
sc.pl.umap(adata, color='leiden')
